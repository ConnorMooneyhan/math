\begin{ex}
	A nonempty family of sets $\mathscr{R} \in \mathscr{P}(X)$ is called a \textbf{ring} if it is closed under finite unions and differences (i.e., if $E_1, \dots, E_n \in \mathscr{R}$, then $\cup _{1}^{n} E_j \in \mathscr{R}$, and if $E, F \in \mathscr{R}$, then $E \setminus F \in \mathscr{R}$).
	A ring that is closed under countable unions is called a \textbf{$\sigma$-ring}.
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item Rings (resp. $\sigma$-rings) are closed under finite (resp. countable) intersections.
		\item If $\mathscr{R}$ is a ring (resp. $\sigma$-ring), then $\mathscr{R}$ is an algebra (resp. $\sigma$-algebra) iff $X \in \mathscr{R}$.
		\item If $\mathscr{R}$ is a $\sigma$-ring, then $\left\lbrace E \subset X : E \in \mathscr{R}\ \mathrm{or}\ E^c \in \mathscr{R} \right\rbrace$ is a $\sigma$-algebra.
		\item If $\mathscr{R}$ is a $\sigma$-ring, then $\left\lbrace E \subset X : E \cap F \in \mathscr{R} \ \mathrm{for\ all} \ F \in \mathscr{R} \right\rbrace$ is a $\sigma$-algebra.
	\end{enumerate}
\end{ex}
\begin{proof}
	\ \begin{enumerate}[label=\textbf{\alph*.}]
		\item If $\mathscr{R}$ is a $\sigma$-ring and $\left\lbrace E_n \right\rbrace_{n \in \mathbb{N}} \subset \mathscr{R}$, then \begin{equation*}
			      \bigcap_{n \in \mathbb{N}} E_n = \left( \bigcup_{n \in \mathbb{N}} E_n \right) \setminus \left( \bigcup \left\lbrace E_a \setminus E_b : a, b \in \mathbb{N} \right\rbrace \right) \tag{1}
		      \end{equation*}
		      Considering the left and right components of the difference in the right hand side of (1), the left is clearly a countable union of members of $\mathscr{R}$, and the right can be seen to be a countable union by identifying each $E_a \setminus E_b$ with $\frac{a}{b}$.
		      Thus the right hand side of (1) is a difference of countable unions of members of $\mathscr{R}$, so it is in $\mathscr{R}$.

		      If $\mathscr{R}$ is a ring, the argument above applies by making $\left\lbrace E_n \right\rbrace$ a finite set.

		\item In either case, if $\mathscr{R}$ is a ring (resp. $\sigma$-ring) and an algebra (resp. $\sigma$-algebra), then given $E \in \mathscr{R}$, $X = E \cup E^c \in \mathscr{R}$.

		      Conversely, if $\mathscr{R}$ is a ring (resp. $\sigma$-ring) and $X \in \mathscr{R}$, then given $E \in \mathscr{R}$, $E^c = X \setminus E \in \mathscr{R}$, so $\mathscr{R}$ is closed under complements.
		      $\mathscr{R}$ is already closed under finite (resp. countable) unions, so $\mathscr{R}$ is an algebra.

		\item Let $A = \left\lbrace E \subset X : E \in \mathscr{R}\ \mathrm{or}\ E^c \in \mathscr{R} \right\rbrace$.
		      Given $E \in A$, by definition $E^c \in A$ since $(E^c)^c = E \in A$.

		      Let $\left\lbrace E_n \right\rbrace_{n \in \mathbb{N}} \subset A$, and express this set as the union of two sets, namely \begin{equation*}
			      \left\lbrace E_n \right\rbrace_{n \in \mathbb{N}} = \left\lbrace F_j \right\rbrace_{j \in J} \cup \left\lbrace G_k \right\rbrace_{k \in K}
		      \end{equation*}
		      where $\left\lbrace F_j \right\rbrace_{j \in J} \subset A$ is the collection of all sets in $A$ who are members of $\mathscr{R}$ and $\left\lbrace G_k \right\rbrace_{k \in K} \subset A$ is the collection of sets in $A$ who are not themselves in $\mathscr{R}$, but whose complements are.
		      Then
		      \begin{align*}
			      \bigcup_{n \in \mathbb{N}} E_n & = \bigcup_{j \in J} F_j \cup \bigcup_{k \in K} G_k                            \\
			                                     & = \bigcup_{j \in J} F_j \cup \left( \bigcap_{k \in K} G_k^c \right)^c \tag{1}
		      \end{align*}
		      Since each $F_j$ is in $\mathscr{R}$, $\bigcup_{j \in J} F_j \in \mathscr{R} \subset A$.
		      Also, we know from (a) that $\sigma$-rings are closed under countable intersections, so since $G_k^c \in R$ for each $G_k$, we have $\bigcap_{k \in K}G_k^c \in R$.
		      Its complement is also in $R$ since $R$ is closed under complementation.
		      Thus (2) is a union of two members of $R$, and is therefore in $R \subset A$.

		\item Let $A = \left\lbrace E \subset X : E \cap F \in \mathscr{R} \ \mathrm{for\ all} \ F \in \mathscr{R} \right\rbrace$.
		      Note that $\mathscr{R} \subset A$ since $\mathscr{R}$ is closed under finite intersections.
		      Given any $E, F \in \mathscr{R}$, we have \begin{align*}
			      E^c \cap F = \left( X \setminus E \right) \cap F = F \setminus E
		      \end{align*}
		      which is a difference in $\mathscr{R}$ and is thus in $\mathscr{R} \subset A$.

		      Given $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset A$ and arbitrary $F \in A$, \begin{equation*}
			      \left( \bigcup_{j \in \mathbb{N}} E_j \right) \cap F = \bigcup_{j \in \mathbb{N}} \left( E_j \cap F \right),
		      \end{equation*}
		      which is a countable union of elements in $\mathscr{R}$.
		      Thus since the choice of $F$ was arbitrary, $\bigcup \left\lbrace E_j \right\rbrace \in A$.
	\end{enumerate}
\end{proof}

\begin{ex}
	Let $\mathscr{M}$ be an infinite $\sigma$-algebra.
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item $\mathscr{M}$ contains an infinite sequence of disjoint nonempty sets.
		\item card($\mathscr{M}$) $\geq c$.
	\end{enumerate}
\end{ex}
\begin{proof}
	\ \begin{enumerate}[label=\textbf{\alph*.}]
    \item Start with $E_1, E_2 \in \mathscr{M}$, both not $X$.
      Take $E_3 \in \mathscr{M}$ not a subset of $E_1 \cup E_2$ and not $X$.
      Continue choosing $E_i$ such that $E_i \neq X$ and $E_i \not\subset \cup_{j = 1}^{i -1} E_j$ for all $i \in \mathbb{Z}^+$.
      Assume that for some $i$, such a choice is impossible.
      Then since $\mathscr{M}$ is infinite, there must be infinitely many subsets of $\cup_{j = 1}^{i-1}E_j$ in $\mathscr{M}$.
      Each of these subsets has a complement which is not a subset of $E_1 \cup E_2$, contradicting the assumption of impossibility.

      Finally, the sequence $\left\lbrace F_i \right\rbrace$, where $F_i = E_i \setminus \cup_{j = 1}^{i -1}E_j$ is a sequence of disjoint members of $\mathscr{M}$ as desired.
		\item Choose a sequence $\left\lbrace E_j \right\rbrace _{1}^{\infty}$ of disjoint nonempty sets in $\mathscr{M}$ (allowable by part (a) of this exercise).
		      Define $f: \mathscr{P}(\mathbb{N}) \to \mathscr{M}$ by $f: S \mapsto \bigcup_{k \in S} E_k$.
		      This is well-defined since $f(S)$ is a countable union of members of $\mathscr{M}$, so $f(S) \in \mathscr{M}$.
		      Also, $f$ is an injection since all the $E_j$ are disjoint.
		      This brings us to the conclusion that since $\mathrm{im }\,f \subset \mathscr{M}$, \begin{equation*}
			      \mathrm{card}\, \mathscr{M} \geq \mathrm{card}(\mathrm{im }\,f) = \mathrm{card}(\mathscr{P}(\mathbb{N})) = c.
		      \end{equation*}
	\end{enumerate}
\end{proof}

\begin{ex}
	An algebra $\mathscr{A}$ is a $\sigma$-algebra iff $\mathscr{A}$ is closed under countable increasing unions (i.e., if $\left\lbrace E_j \right\rbrace _{1}^{\infty} \subset \mathscr{A}$ and $E_1 \subset E_2 \subset \cdots,$ then $\bigcup _{1}^{\infty}E_j \in \mathscr{A}$).
\end{ex}
\begin{proof}
	If $\mathscr{A}$ is a $\sigma$-algebra, then $\mathscr{A}$ is closed under countable unions in general, so it is certainly closed under countable increasing unions in particular.
	On the other hand, if $\mathscr{A}$ is an algebra which is closed under countable increasing unions, then take any $\left\lbrace E_j \right\rbrace _{1}^{\infty} \subset \mathscr{A}$.
	Consider that \begin{equation*}
		\bigcup _{j = 1}^{\infty} E_j = \bigcup _{j = 1}^{\infty} \bigcup _{k = 1}^{j} E_k. \tag{1}
	\end{equation*}
	Clearly, for any $j \in \mathbb{N}$, $\left( \bigcup _{k = 1}^{j} E_k  \right)\subset \left( \bigcup _{k = 1}^{j + 1} E_k \right)$, and each $\bigcup_{k = 1}^{j}E_k$ is a finite union of elements of the algebra $\mathscr{A}$, and thus is in $\mathscr{A}$.
	Then the right hand side of (1) is a countable increasing union of members of $\mathscr{A}$.
	Thus (1) $\in \mathscr{A}$.
\end{proof}

\begin{ex}
	If $\mathscr{M}$ is the $\sigma$-algebra generated by $\mathscr{E}$, then $\mathscr{M}$ is the union of the $\sigma$-algebras generated by $\mathscr{F}$ as $\mathscr{F}$ ranges over all countable subsets of $\mathscr{E}$.
\end{ex}
\begin{proof}
	Let $F$ be the set of all countable subsets of $\mathscr{E}$.
	By Lemma 1.1, since $\mathscr{F} \subset \mathscr{E} \subset \mathscr{M}(\mathscr{E})$ for each $\mathscr{F} \in F$, $\mathscr{M}(\mathscr{F}) \subset \mathscr{M}(\mathscr{E})$.
	Their union, then, is also a subset of $\mathscr{E}$.

	We need to show, then, that $\mathscr{M}(\mathscr{E}) \subset \bigcup_{\mathscr{F} \in F}\mathscr{M}(\mathscr{F})$.
	If we can show that $\bigcup_{\mathscr{F} \in F} \mathscr{M}(\mathscr{F}) = \mathscr{M}(\bigcup_{\mathscr{F} \in F} \mathscr{F})$, then we will be done because $\mathscr{E} \subset \bigcup F$, so $\mathscr{M}(\mathscr{E}) \subset \mathscr{M}(\bigcup F)$.
	That is where we now turn our focus.

	Clearly $\mathscr{M}(\bigcup F) \subset \bigcup_{\mathscr{F} \in F} \mathscr{M}(\mathscr{F})$ since $\bigcup F \subset \bigcup_{\mathscr{F} \in F} \mathscr{M}(\mathscr{F})$, so we need to prove the other direction.
	Given $E \in \bigcup \mathscr{M}(F)$, fix some $\mathscr{F} \in F$ for which $E \in \mathscr{M}(\mathscr{F})$.
	Then since $\mathscr{F} \subset \bigcup F$, $\mathscr{M}(\mathscr{F}) \subset \mathscr{M}(\bigcup F)$, so $E \in \mathscr{M}(\bigcup F)$.
	As observed above, this completes our proof.
\end{proof}

\begin{proof}
	First we verify that $\overline{\mu}$ is a complete measure on $\overline{\mathscr{M}}$.
	Since we can write $\emptyset = \emptyset \cup \emptyset$ where $\emptyset \in \mathscr{M}, \mathscr{N}$, we have $\overline{\mu} = \mu(\emptyset) = 0$.
	Given a disjoint collection $\left\lbrace E_j \cup F_j \right\rbrace_{j \in \mathbb{N}} \subset \overline{\mathscr{M}}$ where each $E_j \in \mathscr{M}$ and each $F_j \subset N_j \in \mathscr{N}$, we have \begin{align*}
		\overline{\mu} \left( \bigcup _{1}^{\infty} \left( E_j \cup F_j \right) \right) & = \overline{\mu} \left( \bigcup _{1}^{\infty} E_j \cup \bigcup _{1}^{\infty} F_j \right) \\
		                                                                                & = \mu \left( \bigcup _{1}^{\infty}E_j \right)                                            \\
		                                                                                & = \sum _{1}^{\infty} \mu \left( E_j \right)                                              \\
		                                                                                & = \sum _{1}^{\infty} \overline{\mu} \left( E_j \cup F_j \right)
	\end{align*}
	where the second equality holds because $\bigcup _{1}^{\infty} E_j \in \mathscr{M}$ and $\bigcup _{1}^{\infty}F_j \in \mathscr{N}$ since both $\mathscr{M}$ and $\mathscr{N}$ are closed under countable unions.
	Of course, since $\overline{\mu}$ is a measure and its domain is complete, $\overline{\mu}$ is a complete measure.

	Now we turn our attention to the uniqueness of $\overline{\mu}$ as an extension of $\mu$ over $\overline{\mathscr{M}}$.
	Assume that $f$ is a measure over $\overline{\mathscr{M}}$ which extends $\mu$.
	Then given $M \cup F \in \overline{\mathscr{M}}$ where $M \in \mathscr{M}$ and $F \subset N \in \mathscr{N}$, by monotonicity we have that $f(F) \leq f(N) = 0$, so $f(F) = 0$.
	We have \begin{equation*}
		f(M) \leq f \left( M \cup F \right) \leq f(M) + f(F) = f(M).
	\end{equation*}
	Thus $f(M \cup F) = f(M) = \mu(M) = \overline{\mu}(M \cup F)$.
\end{proof}

		                                                                                       & = \sum _{j = 1}^{\infty} \sum _{k = 1}^{n} a_k\mu_k (E_j).
	\end{align*}
\end{proof}

\begin{ex}

		                 & = \adjustlimits\sup_{k \geq 1} \inf_{j \geq k} \mu(E_j)                      \\
		                 & = \liminf\mu(E_j)
	\end{align*}
\end{proof}

\begin{ex}

\begin{proof}
	\begin{align*}
		\mu(E \cup F) & = \mu((E \setminus F) \cup (E \cap F) \cup (F \setminus E))                                   \\
		              & = \mu(E \setminus F) + \mu(E \cap F) + \mu(F \setminus E)                                     \\
		              & = \mu(E \setminus F) + \mu(E \cap F) - \mu(E \cap F) + \mu(E \cap F) + \mu(F \setminus E)     \\
		              & = \mu((E \setminus F) \cup (E \cap F)) - \mu(E \cap F) + \mu((E \cap F) \cup (F \setminus E)) \\
		              & = \mu(E) - \mu(E \cap F) + \mu(F).
	\end{align*}
	Thus, $\mu(E \cup F) + \mu(E \cap F) = \mu(E) + \mu(F)$ as desired.
\end{proof}

\begin{ex}
	A finitely additive measure $\mu$ is a measure iff it is continuous from below as in Theorem 1.8c.
	If $\mu(X) < \infty$, $\mu$ is a measure iff it is continuous from above as in Theorem 1.8d.
\end{ex}
\begin{proof}
	For both assertions, if $\mu$ is a measure, then clearly it is continuous from above and below so we restrict ourselves to proving the other direction in either case.
	Also, $\mu(\emptyset) = 0$ since $\mu$ is a finitely additive measure in both cases, so we will only prove countable additivity.

	Given that $\mu$ is a finitely additive measure and it is continuous from below, consider a disjoint collection $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \mathscr{M}$.
	Consider the sequence defined by $F_j = \bigcup _{k = 1}^{j} E_k$, which is clearly an ascending chain.
	Then \begin{align*}
		\mu \left( \bigcup _{j = 1}^{\infty}E_j \right) & = \mu \left( \bigcup _{j = 1}^{\infty} F_j \right)                \\
		                                                & = \lim_{j \to \infty} \mu(F_j)                                    \\
		                                                & = \lim_{j \to \infty} \mu \left( \bigcup _{k = 1}^{j} E_k \right) \\
		                                                & = \lim_{j \to \infty} \sum _{k = 1}^{j} \mu(E_k)                  \\
		                                                & = \sum _{j = 1}^{\infty} \mu \left( E_j \right).
	\end{align*}

	Similarly, assuming that $\mu(X) < \infty$ and $\mu$ is continuous from above, define $\left\lbrace F_j \right\rbrace$ as above.
	Observe that $\left\lbrace F_j^C \right\rbrace$ is a descending chain, so \begin{align*}
		\mu \left( \bigcup _{j = 1}^{\infty}E_j \right) & = \mu \left( \bigcup _{j = 1}^{\infty} F_j \right)                                 \\
		                                                & = \mu \left( \left( \bigcap _{j = 1}^{\infty} F_j^C \right)^C \right)              \\
		                                                & = \mu \left( X \setminus \left( \bigcap _{j = 1}^{\infty} F_j^C \right) \right)    \\
		                                                & = \mu \left( X \right) - \mu \left( \bigcap _{j = 1}^{\infty} F_j^C \right)        \\
		                                                & = \mu \left( X \right) - \lim_{j \to \infty} \mu(F_j^C)                            \\
		                                                & = \mu \left( X \right) - \lim_{j \to \infty} \mu(X \setminus F_j)                  \\
		                                                & = \mu \left( X \right) - \lim_{j \to \infty} (\mu(X) - \mu(F_j))                   \\
		                                                & = \mu \left( X \right) - \lim_{j \to \infty} \mu(X) + \lim_{j \to \infty} \mu(F_j) \\
		                                                & = \lim_{j \to \infty} \mu(F_j)                                                     \\
		                                                & = \sum _{j = 1}^{\infty} \mu \left( E_j \right).
	\end{align*}
\end{proof}

\begin{ex}
	Let $(X, \mathscr{M}, \mu)$ be a finite measure space.
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item If $E, F \in \mathscr{M}$ and $\mu(E \bigtriangleup F) = 0$, then $\mu(E) = \mu(F)$.
		\item Say that $E \sim F$ if $\mu(E \bigtriangleup F) = 0$; then $\sim$ is an equivalence relation on $\mathscr{M}$.
		\item For $E, F \in \mathscr{M}$, define $\rho (E, F) = \mu(E \bigtriangleup F)$.
		      Then $\rho(E, G) \leq \rho(E, F) + \rho(F, G)$, and hence $\rho$ defines a metric on the space $\mathscr{M} / \sim$ of equivalence classes.
	\end{enumerate}
\end{ex}
\begin{proof}\
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item Observe that \begin{align*}
			      0 = \mu(E \bigtriangleup F) = \mu(E \cup F) - \mu(E \cap F),
		      \end{align*}
		      so \begin{align*}
			      \mu(E \cap F) & = \mu(E \cup F)               \\
			                    & = \mu((E \setminus F) \cup F) \\
			                    & = \mu(E \setminus F) + \mu(F) \\
			                    & = \mu (F)
		      \end{align*}
		      since $E \setminus F \subset E \bigtriangleup F$, and thus $\mu (E \setminus F) = 0$ by monotonicity.
		      Swapping $E$ and $F$ in the above calculation reveals that $\mu(E \cap F) = \mu(E)$ as well, so $\mu(F) = \mu(E)$.
		\item Surely given $E \in \mathscr{M}$, $\mu(E \bigtriangleup E) = \mu(\emptyset) = 0$, so $E \sim E$.
		      Given also $F \in \mathscr{M}$ such that $E \sim F$, then since $E \bigtriangleup F = F \bigtriangleup E$, $\mu(F \bigtriangleup E) = 0$ so that $F \sim E$.
		      Finally, if we have $G \in \mathscr{M}$ such that $F \sim G$, then notice that \begin{align*}
			      E \bigtriangleup G \subset (E \bigtriangleup F) \cup (F \bigtriangleup G).
		      \end{align*}
		      The easiest way to see this is with a venn diagram (shoot me), but it can also be shown with some slightly painful set theory.
		      From this, we use monotonicity and subadditivity (respectively) to get \begin{align*}
			      \mu(E \bigtriangleup G) & \leq \mu((E \bigtriangleup F) \cup (F \bigtriangleup G)) \\
			                              & \leq \mu(E \bigtriangleup F) + \mu(F \bigtriangleup G)   \\
			                              & = 0,
		      \end{align*}
		      so $\mu(E \bigtriangleup G) = 0$.
		\item By the definition of $\sim$, $\rho(E, F) = 0$ if and only if $E \sim F$.
		      Also, $\rho(E, F) = E \bigtriangleup F = F \bigtriangleup E = \rho(F, E)$, so the problem statement is justified in that we need only prove the triangle inequality.

		      The following is similar to the end of \textbf{b}, but is reproduced here without the specific choices of $E, F, G$ used in \textbf{b} and adapted for the problem at hand.
		      Given \textit{any} $E, F, G \in \mathscr{M}$, we have \begin{align*}
			      \rho(E, G) & = \mu(E \bigtriangleup G)                                \\
			                 & \leq \mu((E \bigtriangleup F) \cup (F \bigtriangleup G)) \\
			                 & \leq \mu(E \bigtriangleup F) + \mu(F \bigtriangleup G)   \\
			                 & = \rho(E, F) + \rho(F, G)
		      \end{align*}
		      with the second line following from monotonicity and the third from subadditivity.
	\end{enumerate}
\end{proof}

\begin{ex}
	Every $\sigma$-finite measure is semifinite.
\end{ex}
\begin{proof}
	Given a $\sigma$-finite measure space $(X, \mathscr{M}, \mu)$, if $\mu$ is finite, then semifiniteness is obvious.
	Assume, then, that $\mu(X) = \infty$, and let $E \in \mathscr{M}$ such that $\mu(E) = \infty$.
	Also, choose $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \mathscr{M}$ such that $\mu(E_j) < \infty$ for each $j$ and $\bigcup _{1}^{\infty}E_j = X$.
	Then $\bigcup _{1}^{\infty} (E \cap E_j) = E$ and by monotonicity $\mu(E \cap E_j) \leq \mu(E_j) < \infty$ for all $j$.
	It cannot be the case that each $E \cap E_j$ has measure $0$ since \begin{equation*}
		\infty = \mu \left( \bigcup _{1}^{\infty}(E \cap E_j) \right) \leq \sum _{1}^{\infty}\mu(E \cap E_j),
	\end{equation*}
	so we can choose $k \in \mathbb{N}$ such that $0 < \mu(E \cap E_k) < \infty$ as desired.
\end{proof}

	Let $(X, \mathscr{M}, \mu)$ be the measure space in question.
	Given $E \in \mathscr{M}$ such that $\mu(E) = \infty$, define $A = \left\lbrace F : F \in \mathscr{M}, F \subset E, 0 < \mu(F) < \infty \right\rbrace$ and $B = \left\lbrace \mu(F) : F \in A \right\rbrace$.
	Now assume (contradicting the hypothesis) that $B$ is (finitely) bounded from above.
	Then $B$ has a least upper bound $b$.

	Choose $F_1, F_2, \dots \in A$ such that $\left\lbrace \mu(F_j) \right\rbrace_{j \in \mathbb{N}}$ converges to $b$.
	This is possible because $b$ is a limit point of $B$.
	Then define $G_j = \bigcup_{k = 1}^{j} F_k$ for all $j \in \mathbb{N}$.
	$\left\lbrace G_j \right\rbrace$ is an ascending chain of finitely measured sets, and $G_j \subset E$ for all $j$, so $\bigcup_{j = 1}^\infty G_j \subset E$ and $\mu \left( \bigcup_{j = 1}^{\infty}G_j \right) = \lim_{j \to \infty} \mu(G_j)$.
	The sequence $\left\lbrace \mu(G_j) \right\rbrace$ is monotonically increasing due to monotonicity of $\mu$, and it is bounded from above because $\mu(G_j) \leq b$ for any $j$, so it does converge.
	In fact, $\mu(F_j) \leq \mu(G_j) \leq b$ for any $j$, which means that $\left\lbrace \mu(G_j) \right\rbrace$ is sandwiched between $\left\lbrace \mu(F_j) \right\rbrace$ (which converges to $b$) and the constant sequence $\left\lbrace b \right\rbrace$ (which also converges to $b$), so $\left\lbrace \mu(G_j) \right\rbrace$ must also converge to $b$.
	Thus $\mu \left( \bigcup_{j = 1}^{\infty} G_j \right) = b$.

	Given this, we see that $\mu \left( E \setminus \bigcup_{j = 1}^{\infty} G_j \right) = \infty$, so there is a set $G \subset E \setminus \bigcup_{j = 1}^\infty G_j $ such that $0 < \mu(G) < \infty$.
	Then $H = G \cup \bigcup_{j = 1}^\infty G_j$ is such that $H \subset E$, $0 < \mu(H) < \infty$, and $\mu(H) > b$.
	Thus $H \in A$ and $\mu(H) \in B$, which contradicts our assumption of $b$ being a supremum of $B$.
\end{proof}

\end{proof}

		\item Define $\widetilde{\mu}$ on $\widetilde{\mathscr{M}}$ by $\widetilde{\mu}(E) = \mu(E)$ if $E \in \mathscr{M}$ and $\widetilde{\mu}(E) = \infty$ otherwise.
		      Then $\widetilde{\mu}$ is a saturated measure on $\widetilde{\mathscr{M}}$, called the \textbf{saturation} of $\mu$.
		\item If $\mu$ is complete, so is $\widetilde{\mu}$.
		\item Suppose that $\mu$ is semifinite.
		      For $E \in \widetilde{\mathscr{M}}$, define $\underline{\mu}(E) = \sup \left\lbrace \mu(A) : A \in \mathscr{M}\ \mathrm{and}\ A \subset E \right\rbrace$.
		      Then $\underline{\mu}$ is a saturated measure on $\widetilde{\mathscr{M}}$ that extends $\mu$.

		      Let $\mu_0$ be counting measure on $\mathscr{P}(X_1)$, and define $\mu$ on $\mathscr{M}$ by $\mu(E) = \mu_0(E \cap X_1)$.
		      Then $\mu$ is a measure on $\mathscr{M}$, $\widetilde{\mathscr{M}} = \mathscr{P}(X)$, and in the notation of parts (c) and (e), $\widetilde{\mu} \neq \underline{\mu}$.
	\end{enumerate}
\end{ex}
\begin{proof}\
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item Let $E \subset X$ be locally measurable and let $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \mathscr{M}$ such that $\mu(E_j) < \infty$ for each $j$ and $\bigcup_1^\infty E_j = X$.
		      Then $E = \bigcup_1^\infty (E_j \cap E)$, and because $E$ is locally measurable, each $E_j \cap E$ is measurable, so $E$ is a countable union of measurable sets.
		      Thus $E$ is measurable.
		\item Given $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \widetilde{\mathscr{M}}$ and $A \in \mathscr{M}$ such that $\mu(A) < \infty$, \begin{align*}
			      \left( \bigcup_1^\infty E_j \right) \cap A = \bigcup_1^\infty (E_j \cap A),
		      \end{align*}
		      so $\left( \bigcup_1^\infty E_j \right) \cap A$ is the countable union of measurable sets and is thus measurable itself.

		      Also, given $E \in \widetilde{\mathscr{M}}$ and $A$ as before, \begin{equation*}
			      E^C \cap A = A \setminus \left( E \cap A \right).
		      \end{equation*}
		      Since $E \cap A \in \mathscr{M}$ and $\mathscr{M}$ is closed under differences, $E^C \cap A \in \mathscr{M}$.

		      Thus $\widetilde{\mathscr{M}}$ is a $\sigma$-algebra.
		\item First we show that $\widetilde{\mu}$ is a measure.
		      Since $\emptyset \in \mathscr{M}$, $\widetilde{\mu}(\emptyset) = \mu(\emptyset) = 0$.
		      Also, let $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \widetilde{\mathscr{M}}$ be a disjoint collection.
		      If $\bigcup_1^\infty E_j \in \mathscr{M}$, then since each $E_j$ is locally measurable, $E_k = E_k \cap \bigcup_1^\infty E_j \in \mathscr{M}$ for each $k$.
		      Then \begin{align*}
			      \widetilde{\mu}\left( \bigcup_1^\infty E_j \right) & = \mu\left( \bigcup_1^\infty E_j \right) \\
			                                                         & = \sum_1^\infty \mu(E_j)                 \\
			                                                         & = \sum_1^\infty \widetilde{\mu}(E_j).
		      \end{align*}
		      On the other hand, if $\bigcup_1^\infty E_j \notin \mathscr{M}$, then for some $k \in \mathbb{N}$, $E_k \notin \mathscr{M}$.
		      Thus \begin{align*}
			      \widetilde{\mu}\left( \bigcup_1^\infty E_j \right) & = \infty                             \\
			                                                         & = \sum_1^\infty \widetilde{\mu}(E_j)
		      \end{align*}
		      since $\widetilde{\mu}(E_k) = \infty$.

		      Now we show that $\widetilde{\mu}$ is saturated.
		      Let $E \in X$ locally measurable over $\widetilde{\mu}$.
		      We want to show that $E \in \widetilde{\mathscr{M}}$.
		      Let $A \in \mathscr{M}$ such that $\mu(A) < \infty$.
		      Then since $A \in \widetilde{\mathscr{M}}$, $E \cap A \in \widetilde{\mathscr{M}}$, and by monotonicity \begin{equation*}
			      \widetilde{\mu}(E \cap A) < \widetilde{\mu}(A) = \mu(A) < \infty.
		      \end{equation*}
		      Thus since $\widetilde{\mu}(E \cap A)$ is finite, $E \cap A \in \mathscr{M}$.
		      Therefore $E \in \widetilde{\mathscr{M}}$.
		\item Let $F \subset N$ for some $N \in \widetilde{\mathscr{M}}$ where $\widetilde{\mu}(N) = 0$.
		      Then $N \in \mathscr{M}$ since $\widetilde{\mu}(N)$ is finite.
		      Since $\mu$ is complete, $F \in \mathscr{M} \subset \widetilde{\mathscr{M}}$.
		      Thus $\widetilde{\mu}$ is complete.
		\item First we prove that $\underline{\mu}$ is a measure.
		      Since $\emptyset$ is the only subset of $\emptyset$ and $\mu(\emptyset) = 0$, $\underline{\mu}(\emptyset) = \sup \left\lbrace 0 \right\rbrace = 0$.
		      Also, given a disjoint collection $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \widetilde{\mathscr{M}}$, \begin{align*}
			      \underline{\mu}\left( \bigcup_1^\infty E_j \right) & = \sup \left\lbrace \mu(A) : A \in \mathscr{M}\ \mathrm{and}\ A \subset \bigcup_1^\infty E_j \right\rbrace                                                \\
			                                                         & = \sup \left\lbrace \mu \left( \bigcup_{k = 1}^\infty A \cap E_k \right)  : A \in \mathscr{M}\ \mathrm{and}\ A \subset \bigcup_1^\infty E_j \right\rbrace \\
			                                                         & = \sup \left\lbrace \sum_{k = 1}^\infty \mu \left( A \cap E_k \right) : A \in \mathscr{M}\ \mathrm{and}\ A \subset \bigcup_1^\infty E_j \right\rbrace     \\
			                                                         & = \sum_{k = 1}^\infty \sup \left\lbrace \mu \left( A \cap E_k \right) : A \in \mathscr{M}\ \mathrm{and}\ A \subset \bigcup_1^\infty E_j \right\rbrace     \\
			                                                         & = \sum_{k = 1}^\infty \sup \left\lbrace \mu(A) : A \in \mathscr{M}\ \mathrm{and}\ A \subset E_k \right\rbrace                                             \\
			                                                         & = \sum_{k = 1}^\infty \underline{\mu}(E_k).
		      \end{align*}
		      Thus $\underline{\mu}$ is a measure.

		      To show that it extends $\mu$ is easy since by monotonicity, any subset of some $E \in \mathscr{M}$ has measure less than or equal to that of $E$, so \begin{equation*}
			      \underline{\mu}(E) = \sup \left\lbrace \mu(A) : A \in \mathscr{M}\ \mathrm{and}\ A \subset E \right\rbrace = \mu(E).
		      \end{equation*}
		      Now we just need to show saturation.
		      In particular, if $E \in X$ is locally measurable over $\underline{\mu}$, then we need to show that $E \in \widetilde{\mathscr{M}}$.
		      Thus we take $A \in \mathscr{M}$ such that $\mu(A) < \infty$ and see that since $A \in \widetilde{\mathscr{M}}$, $E \cap A \in \widetilde{\mathscr{M}}$.
		      Then $E \cap A$ is locally measurable over $\mu$, so $E \cap A = (E \cap A) \cap A \in \mathscr{M}$.
		      Therefore $\underline{\mu}$ is saturated.
		\item First we show that $\mu$ is a measure on $\mathscr{M}$.
		      We see that $\mu(\emptyset) = \mu_0(\emptyset \cap X_1) = \mu_0(\emptyset) = 0$.
		      Let $\left\lbrace E_j \right\rbrace_{j \in \mathbb{N}} \subset \mathscr{M}$ be disjoint.
		      Then \begin{align*}
			      \mu \left( \bigcup_1^\infty E_j \right) & = \mu_0 \left( \left( \bigcup_1^\infty E_j \right) \cap X_1 \right) \\
			                                              & = \mu_0 \left( \bigcup_1^\infty \left( E_j \cap X_1 \right) \right) \\
			                                              & = \sum_1^\infty \mu_0(E_j \cap X_1)                                 \\
			                                              & = \sum_1^\infty \mu(E_j).
		      \end{align*}

		      Next we want to show that $\widetilde{\mathscr{M}} = \mathscr{P}(X)$, or in other words, that every subset of $X$ is locally measurable.
		      Let $E \subset X$ and $A \in \mathscr{M}$ such that $\mu(A) < \infty$.
		      Notice that $A \cap X_1$ must be finite since $\mu_0(A \cap X_1) < \infty$.
		      Thus $A^C$ is uncountable because $X_1$ is uncountable and $A^C \cap X_1$ = $X_1 \setminus \left( A \cap X_1 \right)$.
		      This means $A$ cannot be co-countable, so since $A \in \mathscr{M}$, it must be countable.
		      Seeing this, it is obvious that $E \cap A$ is countable because $E \cap A \subset A$, so $E \cap A \in \mathscr{M}$.

		      Finally, to see that $\widetilde{\mu} \neq \underline{\mu}$, observe that $X_2 \notin \mathscr{M}$ because $X_2$ is uncountable and $X_2^C = X_1$ is uncountable.
		      Then $\widetilde{\mu}(X_2) = \infty$ because $X^2 \notin \mathscr{M}$, but $\mu(X_2) = \mu_0(X_2 \cap X_1) = \mu_0(\emptyset) = 0$, so any subset of $X_2$ also has measure 0.
		      Thus \begin{equation*}
			      \underline{\mu}(X_2) = \sup \left\lbrace \mu(A) : A \in \mathscr{M}\ \mathrm{and}\ A \subset X_2 \right\rbrace = \sup \left\lbrace 0 \right\rbrace = 0.
		      \end{equation*}
		      We conclude that since $\widetilde{\mu}(X_2) = \infty \neq 0  = \underline{\mu}(X_2)$, $\widetilde{\mu} \neq \underline{\mu}$.
	\end{enumerate}
\end{proof}

\begin{ex}
	If $\mu^*$ is an outer measure on $X$ and $\left\lbrace A_j \right\rbrace _{1}^{\infty}$ is a sequence of disjoint $\mu^*$-measurable sets, then $\mu^*(E \cap (\bigcup_1^\infty A_j)) = \sum_1^\infty \mu^*(E \cap A_j)$ for any $E \subset X$.
\end{ex}
\begin{proof}
	Notice that \begin{equation*}
		\mu^* \left( E \cap \left( \bigcup_1^\infty A_j \right) \right) = \mu^* \left( \bigcup_1^\infty E \cap A_j \right) \leq \sum_1^\infty \mu^* \left( E \cap A_j \right),
	\end{equation*}
	so to prove equality, we need only show that $\mu^* \left( E \cap \bigcup_1^\infty A_j \right) \geq \sum_1^\infty \mu^* \left( E \cap A_j \right)$.

	Given $n \in \mathbb{N}$, define $E_n = E \cap A_n$ and $B_n = \bigcup_{j = 1}^n A_j$.
	Then \begin{align*}
		\mu^*(E \cap B_n) & = \mu^*(E \cap B_n \cap A_n) + \mu^*(E \cap B_n \cap A_n^C) \\
		                  & = \mu^*(E \cap A_n) + \mu^*(E \cap B_{n - 1})
	\end{align*}
	when $n > 1$.
	If $n = 1$, then clearly $\mu^*(E \cap B_1) = \mu^*(E \cap A_1)$ by definition.
	By induction, we get $\mu^*(E \cap B_n) = \sum_{j = 1}^n \mu^*(E \cap A_j)$.
	Thus \begin{equation*}
		\lim_{n \to \infty} \mu^*(E \cap B_n) = \sum_{j = 1}^\infty \mu^*(E \cap A_j). \tag{1}
	\end{equation*}
	Since each $E \cap B_n \subset E \cap \bigcup_{j = 1}^\infty A_j$, each $\mu^*(E \cap B_n) \leq \mu^*(E \cap \bigcup_{j = 1}^\infty A_j)$, so we see that \begin{equation*}
		\mu^*(E \cap \bigcup_{j = 1}^\infty A_j) \geq \lim_{n \to \infty} \mu^*(E \cap B_n).
	\end{equation*}
	Combining with (1) yields $\mu^*(E \cap \bigcup_{j = 1}^\infty A_j) \geq \sum_{j = 1}^\infty \mu^*(E \cap A_j)$.
\end{proof}

		      (It suffices to assume $E \subset [0, 1]$. In the notation of \S 1.1, $E = \bigcup_{r \in R}E \cap N_r$.)
	\end{enumerate}
\end{ex}
\begin{proof}\
	\begin{enumerate}[label=\textbf{\alph*.}]
		\item Given $r \in R$, define $E_r$ to be \begin{equation*}
			      E_r = \left\lbrace x + r : x \in E \cap [0, 1 - r) \right\rbrace \cup \left\lbrace x + r - 1 : x \in E \cap [1 - r, 1) \right\rbrace.
		      \end{equation*}
		      Immediately, we can see that \begin{align*}
			      m(E_r) & = m \left( \left\lbrace x + r : x \in E \cap [0, 1 - r) \right\rbrace \cup \left\lbrace x + r - 1 : x \in E \cap [1 - r, 1) \right\rbrace \right)              \\
			             & = m \left( \left\lbrace x + r : x \in E \cap [0, 1 - r) \right\rbrace \right) + m\left( \left\lbrace x + r - 1 : x \in E \cap [1 - r, 1) \right\rbrace \right) \\
			             & = m \left( \left\lbrace x : x \in E \cap [0, 1 - r) \right\rbrace \right) + m\left( \left\lbrace x : x \in E \cap [1 - r, 1) \right\rbrace \right)             \\
			             & = m \left( \left\lbrace x : x \in E \cap [0, 1 - r) \right\rbrace \cup \left\lbrace x : x \in E \cap [1 - r, 1) \right\rbrace \right)                          \\
			             & = m \left( E \right),
		      \end{align*}
		      where the third equality comes by the translation invariance of $m $.
		      Using a similar argument to that of \S 1.1, we see that \begin{equation*}
			      m \left(\bigcup_{r \in R} E_r \right) \leq m\left([0, 1) \right) = 1,
		      \end{equation*}
		      but \begin{equation*}
			      m \left(\bigcup_{r \in R} E_r \right) = \sum_{r \in R} m \left( E_r \right) = \sum_{r \in R} m(E).
		      \end{equation*}
		      Clearly, the final expression above cannot be finite unless $m(E) = 0$, so since it is forced to be $\leq 1$ (i.e. finite), in fact $m(E) = 0$.
		\item Since $X = \bigcup_{z \in \mathbb{Z}} I_z$ where each $I_z = [z, z+1)$, \begin{equation*}
			      0 < m(E) = m \left( \bigcup_{z \in \mathbb{Z}} I_z \cap E \right) = \sum_{z \in \mathbb{Z}} m(I_z \cap E),
		      \end{equation*}
		      so $m(I_j \cap E) > 0$ for some $j \in \mathbb{Z}$.
		      Then \begin{equation*}
			      \left( I_j \cap E \right) - j \subset I_0
		      \end{equation*}
		      and \begin{equation*}
			      m \left( I_j \cap E \right) - j = m \left( I_j \cap E \right) > 0,
		      \end{equation*}
		      so since $I_0 = \bigcup_{r \in R} N_r$, there exists a $k$ such that \begin{equation*}
			      m \left( \left( \left( I_j \cap E \right) - j \right) \cap N_k \right) > 0
		      \end{equation*}
          Denote $\left( \left( I_j \cap E \right) - j \right) \cap N_k$ by $F$.
		      Note that part \textbf{a.} holds for each $N_r$ in \S 1.1 since the choice of $N$ was arbitrary and each $N_r$ consists of exactly one representative from each equivalence class.
          It follows from \textbf{a.}, then, that since $F \subset N_k$ and $m(F) > 0$, $F$ is not Lebesgue measurable.

          Finally, notice that $F + j \subset I_j \cap E \subset E$, so $F + j$ is a nonmeasurable set contained in $E$ since if $F + j$ were measurable, $F$ would be as well.
	\end{enumerate}
\end{proof}
