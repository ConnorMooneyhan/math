\begin{ex}
  Let $H$ be a Hilbert space and define $T: H \to H'$ as $z \mapsto \langle \cdot, z \rangle$.
  Show that: \begin{enumerate}[label=(\arabic*)]
    \item $T$ is isometric, bijective and conjugate linear: $T(ax + by) = \overline{a}T(x) + \overline{b}T(y)$.
    \item $H'$ is a Hilbert space with inner product $\langle T(x), T(y) \rangle = \overline{\langle x, y \rangle} = \langle y, x \rangle$.
      Therefore, if $H$ is a real Hilbert space then it is isomorphic to $H'$.
    \item Any complex Hilbert space $H$ is isomorphic to $(H')'$.
  \end{enumerate}
\end{ex}

\begin{ex}

\end{ex}

\begin{ex}
  Let $X = \mathbb{R}^2$ be equipped with the box norm.
  Let $u = \langle 1, 0 \rangle$ and $T$ be the linear functional on $\mathrm{span}(u)$ satisfying $T(u) = 1$. \begin{enumerate}[label=(\arabic*)]
    \item Show that for any $M > 0$ there exists $v \in \mathbb{R}$, linearly independent with $u$, such that the extension of $T$ determined by $T(v) = 0$ satisfies $||T|| \geq M$.
    \item Find a value of $c$ such that the extension of $T$ determined by $T(-1, 1) = c$ satisfies $||T|| = 1$.
      Is the choice of $c$ unique?
  \end{enumerate}
\end{ex}
