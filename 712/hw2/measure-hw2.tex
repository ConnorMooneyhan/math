\begin{ex}
	(1.A.8)
	Suppose $f: [a, b] \to \mathbb{R}$ is Riemann integrable.
	Prove that \begin{equation}\label{1.1}
		\int_a^b f = \lim_{n \to \infty} \frac{b - a}{n} \sum_{j = 1}^n f \left( a + \frac{j(b-a)}{n} \right).
	\end{equation}
\end{ex}
\begin{proof}
	Define a partition $P_n = \left\lbrace a, a + \frac{b-a}{n}, \dots, a + \frac{(n-1)(b - a)}{n}, b \right\rbrace$ for each $n \in \mathbb{Z}^+$.
	Let $\varepsilon > 0$ and choose $P$ such that $L(f, P, [a, b]) > \int_a^b f - \frac{\varepsilon}{2}$.

	Then \begin{equation}\label{1.2}
		\int_a^b f- \frac{\varepsilon}{2} < L(f, P, [a, b]) \leq L(f, P \cup P_n, [a, b]).
	\end{equation}

  Let $\alpha$ be the number of points in $P$.
  Since $P_n$ and $P \cup P_n$ agree on all subintervals of $P_n$ not containing a point of $P$, their difference is entirely determined by the behavior on subintervals containing a point of $P$, of which there are at most $\alpha$, and each of which has length $\frac{1}{n}$.
  Since $f$ is Riemann integrable, it is bounded, so let $U = \sup f$ and $L = \inf f$.
  Then any infimum along a subinterval is greater than or equal to $L$, and any supremum along a subinterval is less than or equal to $U$.
  Thus we obtain an upper bound on the difference of the lower sums: \begin{equation*}
    L(f, P \cup P_m, [a,b]) - L(f, P_m, [a,b]) \leq \frac{\alpha}{n}(U - L).
  \end{equation*}
  Since $\alpha$ is fixed, we can choose an $m$ large enough such that \begin{align*}
		         & L(f, P \cup P_m, [a, b]) - L(f, P_m, [a, b]) < \frac{\varepsilon}{2}  \\
		\implies & - \frac{\varepsilon}{2} < L(f, P_m, [a, b]) - L(f, P \cup P_m, [a, b]).
	\end{align*}
	Adding this to \eqref{1.2}, we get \begin{equation*}
		\int_a^b f - \varepsilon < L(f, P_m, [a, b]).
	\end{equation*}
	An entirely analagous argument gives us an $m$ such that \begin{equation*}
		\int_a^b f + \varepsilon > U(f, P_m, [a, b]).
	\end{equation*}
	Finally, we observe that since \begin{equation*}
		\inf_{\left[a + \frac{(j-1)(b-a)}{n}, a + \frac{j(b-a)}{n}\right]} f \leq f \left( a + \frac{j(b-a)}{n} \right) \leq \sup_{\left[ a + \frac{(j-1)(b-a)}{n}, a + \frac{j(b-a)}{n} \right]} f,
	\end{equation*}
	\begin{equation*}
		L(f, P_n, [a, b]) \leq \frac{b-a}{n} \sum_{j=1}^n f \left( a + \frac{j(b-a)}{n} \right) \leq U(f, P_n, [a,b]),
	\end{equation*}
	and thus since we showed that the lower and upper sums over $P_n$ create sequences which limit to $\int_a^b f$, the middle sum above is squeezed so that \eqref{1.1} is satisfied.
\end{proof}

\begin{ex}
	(1.A.12)
	Suppose $f: [a, b] \to \mathbb{R}$ is Riemann integrable.
	Prove that $|f|$ is Riemann integrable and that \begin{equation*}
		\left| \int_a^b f \right| \leq \int_a^b |f|.
	\end{equation*}
\end{ex}
\begin{proof}
	We begin by considering the relationship between $\sup_{[x, y]}|f| - \inf_{[x, y]} |f|$ and $\sup_{[x,y]}f - \inf_{[x,y]} f$ for any interval $[x,y]$.
	In particular, we want to prove that $\sup_{[x,y]}|f| - \inf_{[x,y]}|f| \leq \sup_{[x,y]}f - \inf_{[x,y]}f$ \textcolor{red}{(needs proof)}.

	Now, let $\varepsilon > 0$ and choose a partition $P = \left\lbrace a = x_0, x_1, \dots, x_n = b \right\rbrace$ such that \begin{equation*}
		U(f, P, [a, b]) - L(f,P,[a,b]) < \varepsilon.
	\end{equation*}
	Then \begin{align*}
		U(|f|, P, [a, b]) - L(|f|, P, [a,b]) & = \sum_{i=1}^n(x_i-x_{i-1}) \sup_{[x_{i-1},x_i]}|f| - \sum_{j=1}^n(x_j-x_{j-1})\inf_{[x_{j-1},x_j]}|f| \\
		                                     & = \sum_{i=1}^n(x_i - x_{i-1}) \left( \sup_{[x_{i-1},x_i]}|f| - \inf_{[x_{i-1},x_i]} |f| \right)        \\
		                                     & \leq \sum_{i=1}^n(x_i - x_{i-1}) \left( \sup_{[x_{i-1},x_i]}f - \inf_{[x_{i-1},x_i]} f \right)         \\
		                                     & = U(f, P, [a, b]) - L(f, P, [a,b])                                                                     \\
		< \varepsilon.
	\end{align*}
	Thus $|f|$ is Riemann integrable since $\varepsilon$ is arbitrary.

	For the inequality, we want to show the equivalent statement that \begin{equation*}
		-\int_a^b |f| \leq \int_a^b f \leq \int_a^b |f|.
	\end{equation*}
	The second inequality comes from the fact that for any interval $[\alpha, \beta]$ and any partition $P$ of $[a,b]$, \begin{align*}
		         & f(x) \leq |f(x)| \quad \text{(for all $x$)}              \\
		\implies & \inf_{[\alpha, \beta]} f \leq \inf_{[\alpha, \beta]} |f| \\
		\implies & L(f, P, [a, b]) \leq L(|f|, P, [a,b])                    \\
		\implies & L(f, [a,b]) \leq L(|f|, [a,b])                           \\
		\implies & \int_a^b f \leq \int_a^b |f|.
	\end{align*}
  The first inequality comes from the fact that $-\inf_{[x_{i-1}, x_i]} |f| \leq \inf_{[x_{i-1},x_i]} f$ since $-|f(x)| \leq f(x)$ for all $x$, so $L(-|f|, P, [a, b]) \leq L(f, P, [a, b])$.
\end{proof}

\begin{ex}
	(1.B.1)
	Define $f: [0, 1] \to \mathbb{R}$ as follows: \begin{equation*}
		f(a) = \begin{cases}
			0           & \text{if $a$ is irrational},                                                                                                              \\
			\frac{1}{n} & \parbox[t]{.55\textwidth}{if $a$ is rational and $n$ is the smallest positive integer such that $a = \frac{m}{n}$ for some integer $m $.}
		\end{cases}
	\end{equation*}
	Show that $f$ is Riemann integrable and compute $\displaystyle\int_0^1 f$.
\end{ex}
\begin{proof}
	For any partition, the lower Riemann sum is 0 since there is an irrational number contained within every subinterval.
	Thus, we will show that we can find a partition $P_n$ for which $U(f, P_n, [0,1]) < \frac{1}{n}$, allowing us to conclude that $U(f, [0,1]) = 0$.

	We begin with the case $n = 2$.
	Let $\varepsilon > 0$ be small enough that $[\frac{1}{2} - \varepsilon, \frac{1}{2} + \varepsilon]$ does not contain $\frac{m }{3}$ where $0 \leq m \leq 3$ and $\frac{1}{3} + \frac{1}{3}\varepsilon < \frac{1}{2}$.
	Consider the partition \begin{equation*}
		P_2 = \left\lbrace 0, \frac{1}{2} - \varepsilon, \frac{1}{2} + \varepsilon, 1 \right\rbrace.
	\end{equation*}
	Then \begin{align*}
		U(f, P_2, [0, 1]) & = \left( \frac{1}{2} - \varepsilon \right) \cdot \frac{1}{3} + 2\varepsilon \cdot \frac{1}{2} + \left( \frac{1}{2} - \varepsilon \right) \cdot \frac{1}{3} \\
		                  & = \left( \frac{1}{2} - \varepsilon \right) \cdot \frac{2}{3} + \varepsilon                                                                                 \\
		                  & = \frac{1}{3} - \frac{2}{3}\varepsilon + \varepsilon                                                                                                       \\
		                  & = \frac{1}{3} + \frac{1}{3}\varepsilon                                                                                                                     \\
		                  & \leq \frac{1}{2}.
	\end{align*}
	We can follow a similar procedure for a general $P_n$, choosing partition points close enough to $\frac{m }{\ell}$ for all $\ell \leq n$ and $0 \leq m \leq \ell$ so that the upper sum is determined up to some fraction of $\varepsilon$ by the larger subintervals, whose maximum value is at most $\frac{1}{n+1}$.
\end{proof}

\begin{ex}
	(1.B.4)
	Give an example of bounded functions $f, g: [0, 1] \to \mathbb{R}$ such that \begin{equation*}
		L(f, [0, 1]) + L(g, [0, 1]) < L(f + g, [0, 1])
	\end{equation*}
	and \begin{equation*}
		U(f + g, [0, 1]) < U(f, [0, 1]) + U(g, [0, 1]).
	\end{equation*}
\end{ex}
\begin{proof}
	Define \begin{equation*}
		f(x) = \begin{cases}
			1  & \text{if $x \in \mathbb{Q}$}    \\
			-1 & \text{if $x \notin \mathbb{Q}$}
		\end{cases}
	\end{equation*}
	and \begin{equation*}
		g(x) = \begin{cases}
			-1 & \text{if $x \in \mathbb{Q}$}     \\
			1  & \text{if $x \notin \mathbb{Q}$}.
		\end{cases}
	\end{equation*}
	Then $(f + g)(x) = 0$ for all $x \in [0, 1]$, so \begin{equation*}
		L(f, [0, 1]) + L(g, [0, 1]) = -1 - 1 = -2 < 0 = L(f + g, [0, 1])
	\end{equation*}  and \begin{equation*}
		U(f, [0, 1]) + U(g, [0, 1]) = 1 + 1 = 2 > 0 = U(f + g, [0, 1]).
	\end{equation*}
\end{proof}

\begin{ex}
	(2.A.1)
	Prove that if $A$ and $B$ are subsets of $\mathbb{R}$ and $|B| = 0$, then $|A \cup B| = |A|$.
\end{ex}
\begin{proof}
	By countable subadditivity, \begin{equation*}
		|A \cup B| \leq |A| + |B| = |A| + 0 = |A|.
	\end{equation*}
	By monotonicity, since $A \subset A \cup B$, \begin{equation*}
		|A| \leq |A \cup B|.
	\end{equation*}
	Therefore, $|A \cup B| = |A|$.
\end{proof}

\begin{ex}
	(2.A.8)
	Prove that if $A \subset \mathbb{R}$ and $t > 0$, then $|A| = |A \cap (-t, t)| + |A \cap (\mathbb{R} \setminus (-t, t))|$.
\end{ex}
\begin{proof}
	By countable subadditivity, \begin{equation*}
		|A| \leq |A \cap (-t, t)| + |A \cap (\mathbb{R} \setminus (-t, t))|.
	\end{equation*}
	For the other direction, note that \begin{equation*}
		|A| = |A \setminus \left\lbrace -t, t \right\rbrace |
	\end{equation*}
	by 2.A.1 since $|\left\lbrace -t, t \right\rbrace| = 0$.
	Then \begin{align*}
		|A \cap (-t, t)| + |A \cap (\mathbb{R} \setminus (-t, t))| & = |A \cap (-t, t)| + |(A \cap (-\infty, -t]) \cup (A \cap [t, \infty))| \\
		                                                           & = |A \cap (-t, t)| + |(A \cap (-\infty, -t)) \cup (A \cap (t, \infty))| \\
		                                                           & \leq |A \cap(-t, t)| + |A \cap (-\infty, -t)| + |A \cap (t, \infty)|.
	\end{align*}
	Let $\varepsilon > 0$ and let $\left\lbrace I_n \right\rbrace_{n \in \mathbb{Z}}$ be a collection of open intervals such that $A \subset \bigcup \left\lbrace I_n \right\rbrace$ and \begin{equation*}
		|A| + \varepsilon \geq \sum_{n=1}^\infty \ell(I_n).
	\end{equation*}
	Let $J_n = I_n \cap (-\infty, t)$, $K_n = I_n \cap (-t, t)$, and $L_n = I_n \cap (t, \infty)$, each of which is either empty or an open interval.
	For each $n$, $J_n$, $K_n$, and $L_n$ are pairwise disjoint, and each is either an open interval or the empty set (which will not cause problems for our purposes).
	Since $J_n \cup K_n \cup L_n \subset I_n$, \begin{equation*}
		\ell(I_n) \geq \ell(J_n) + \ell(K_n) + \ell(L_n),
	\end{equation*}
	and since \begin{align*}
		 & A \cap (-\infty, -t) \subset \bigcup J_n,       \\
		 & A \cap (-t, t) \subset \bigcup K_n, \text{ and} \\
		 & A \cap (t, \infty) \subset \bigcup L_n,
	\end{align*}
	we have \begin{align*}
		|A| + \varepsilon & \geq \sum_{n=1}^\infty \ell(I_n)                                                                                                 \\
		                  & \geq \sum_{n=1}^\infty \ell(J_n) + \sum_{n=1}^\infty \ell(K_n) + \sum_{n=1}^\infty \ell(L_n)                                     \\
		                  & = \sum_{n=1}^\infty |J_n| + \sum_{n=1}^\infty |K_n| + \sum_{n=1}^\infty |L_n|                                                    \\
		                  & \geq \left| \bigcup_{n=1}^\infty J_n \right| + \left| \bigcup_{n=1}^\infty K_n \right| + \left| \bigcup_{n=1}^\infty L_n \right| \\
		                  & \geq |A \cap (-\infty, -t)| + |A \cap (-t, t)| + |A \cap (t, \infty)|                                                            \\
		                  & = |A \cap (-t, t)| + |A \cap (\mathbb{R} \setminus (-t, t))|.
	\end{align*}
	Since this is true for all $\varepsilon > 0$, we have \begin{equation*}
		|A| \geq |A \cap (-t, t)| + |A \cap (\mathbb{R} \setminus (-t, t))|.
	\end{equation*}
\end{proof}
